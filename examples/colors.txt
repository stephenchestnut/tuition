# Color reference for tuition aliases
# Slide 1: foreground and background colors
printf '%s\n\n' 'Foreground colors'; for row in 'FGBLACK FGRED FGGREEN FGYELLOW' 'FGBLUE FGMAGENTA FGCYAN FGWHITE' 'FGGRAY FGBRED FGBGREEN FGBYELLOW' 'FGBBLUE FGBMAGENTA FGBCYAN FGBWHITE'; do for name in $row; do eval "esc=\$$name"; printf '%b%-12s%b  ' "$esc" "$name" "$RESET"; done; printf '\n'; done; printf '\n%s\n\n' 'Background colors'; for row in 'BGBLACK|FGWHITE BGRED|FGWHITE BGGREEN|FGBLACK BGYELLOW|FGBLACK' 'BGBLUE|FGWHITE BGMAGENTA|FGWHITE BGCYAN|FGBLACK BGWHITE|FGBLACK' 'BGBBLACK|FGWHITE BGBRED|FGBLACK BGBGREEN|FGBLACK BGBYELLOW|FGBLACK' 'BGBBLUE|FGWHITE BGBMAGENTA|FGWHITE BGBCYAN|FGBLACK BGBWHITE|FGBLACK'; do for spec in $row; do name=${spec%%|*}; fgname=${spec#*|}; eval "bg=\$$name"; eval "fg=\$$fgname"; printf '%b%b%-12s%b  ' "$bg" "$fg" "$name" "$RESET"; done; printf '\n'; done
# Slide 2: style modifiers and usage
